# Backend Multi-Molde Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move the backend from one `Molde` per user (`Molde.userId` as primary key) to many moldes per user (`Molde.id` as primary key, ownership-checked), and re-home the pattern/billing API routes under `/api/moldes/[id]/*` — so the dashboard (built in a follow-up plan) has real multiple moldes to list.

**Architecture:** `Molde` gets its own `id` (cuid) and `userId` becomes a plain indexed foreign key. Every DB access that used to key off `userId` alone now takes `(moldeId, userId)` and returns `null`/`false` when the row doesn't exist _or_ belongs to someone else — the two cases are indistinguishable to the caller, so a route just turns `null` into a 404. `withMolde`/`withMoldeAndWallet` keep their lock-then-mutate-then-save shape but return `T | null`. The client-side `useEditorStore` gains a `moldeId` it threads through every API call instead of assuming "the" molde.

**Tech Stack:** Next.js 16 (App Router, Route Handlers with `params: Promise<...>`), Prisma 6 / PostgreSQL, Zustand, Vitest.

## Global Constraints

- Commit messages: Conventional Commits, English subject/body (`CLAUDE.md`).
- Code identifiers/wire keys stay English; no UI copy changes beyond what's listed in Task 13 (`CLAUDE.md`).
- `npm run build` (includes the TypeScript check) and `npm test` must pass before each commit that isn't itself a fixup of the previous step's failure.
- Existing dev data is **not** migrated — the migration in Task 1 drops and recreates `Molde` (same precedent as `prisma/migrations/20260817002232_reset_moldes_english`).
- This plan does **not** touch UI beyond the minimum needed to keep the app functional end-to-end (Topbar's delete button, a bare `/molde/[id]` page, an interim `/` bridge). The real dashboard UI is a separate follow-up plan.

```ts
// lib/editor/thumbnail.ts
import type { Pattern, PieceOutline } from "@/lib/types";

function pieceMarkup(outline: PieceOutline): string {
  if (outline.kind === "path") return `<path d="${outline.d}" />`;
  return `<polygon points="${outline.points}" />`;
}

// Server-built dashboard-card thumbnail from a molde's already-drafted
// pieces — no redraft, no AI call. Blank moldes render an empty <g>.
export function buildThumbnailSvg(pattern: Pattern): string {
  const vb = pattern.viewBox ?? { w: 620, h: 380 };
  const shapes = pattern.pieces.map((p) => pieceMarkup(p.outline)).join("");
  return (
    `<svg viewBox="0 0 ${vb.w} ${vb.h}" xmlns="http://www.w3.org/2000/svg">` +
    `<g fill="#fff" stroke="#1c1c19" stroke-width="1.7">${shapes}</g></svg>`
  );
}
```

```ts
// lib/db/molde.ts
import { NextResponse } from "next/server";
import { Prisma } from "@prisma/client";
import { prisma } from "./prisma";
import { seedMoldeState } from "@/lib/editor/state";
import { buildThumbnailSvg } from "@/lib/editor/thumbnail";
import type { GarmentType, MoldeState } from "@/lib/types";

type Tx = Prisma.TransactionClient;

// The timeout needs to cover the AI call (proposeGarment ~5–12s), which runs
// INSIDE the transaction — Prisma's 5s default would time out. Holding the
// lock during the AI call is intentional: it's what serializes the user's requests.
export const TX_OPTS = { timeout: 30_000, maxWait: 10_000 } as const;

function toState(data: Prisma.JsonValue): MoldeState {
  return data as unknown as MoldeState;
}

export function moldeNotFound(): NextResponse {
  return NextResponse.json(
    { status: "not_found", error: "Molde não encontrado." },
    { status: 404 },
  );
}

// Locks and reads the row, scoped to (id, userId) — a molde owned by someone
// else reads as null, identical to a molde that doesn't exist at all.
// Exported so lib/db/molde-wallet.ts can lock this row inside its own
// combined transaction without duplicating the SQL.
export async function lockMolde(
  tx: Tx,
  moldeId: string,
  userId: string,
): Promise<MoldeState | null> {
  const rows = await tx.$queryRaw<{ data: Prisma.JsonValue }[]>`
    SELECT "data" FROM "Molde" WHERE "id" = ${moldeId} AND "userId" = ${userId} FOR UPDATE`;
  return rows[0] ? toState(rows[0].data) : null;
}

export async function saveMolde(
  tx: Tx,
  moldeId: string,
  state: MoldeState,
): Promise<void> {
  await tx.$executeRaw`
    UPDATE "Molde" SET "data" = ${JSON.stringify(state)}::jsonb, "updatedAt" = now()
    WHERE "id" = ${moldeId}`;
}

// load → fn(state) → save, inside an interactive transaction with a row lock.
// Returns null (without calling fn) when the molde doesn't exist or isn't
// the caller's — routes turn that into moldeNotFound().
export async function withMolde<T>(
  moldeId: string,
  userId: string,
  fn: (state: MoldeState) => T | Promise<T>,
): Promise<T | null> {
  return prisma.$transaction(async (tx) => {
    const state = await lockMolde(tx, moldeId, userId);
    if (!state) return null;
    const result = await fn(state);
    await saveMolde(tx, moldeId, state);
    return result;
  }, TX_OPTS);
}

// Read-only fetch, ownership-scoped. Null when missing or not owned.
export async function loadMolde(
  moldeId: string,
  userId: string,
): Promise<MoldeState | null> {
  const row = await prisma.molde.findUnique({ where: { id: moldeId } });
  if (!row || row.userId !== userId) return null;
  return toState(row.data);
}

export async function createMolde(userId: string): Promise<string> {
  const data = seedMoldeState() as unknown as Prisma.InputJsonValue;
  const row = await prisma.molde.create({ data: { userId, data } });
  return row.id;
}

export interface MoldeListItem {
  id: string;
  name: string;
  garmentType: GarmentType | null;
  updatedAt: string;
  thumbnailSvg: string;
}

export async function listMoldesForUser(
  userId: string,
): Promise<MoldeListItem[]> {
  const rows = await prisma.molde.findMany({
    where: { userId },
    orderBy: { updatedAt: "desc" },
  });
  return rows.map((row) => {
    const state = toState(row.data);
    return {
      id: row.id,
      name: state.pattern.name,
      garmentType: state.pattern.spec?.type ?? null,
      updatedAt: row.updatedAt.toISOString(),
      thumbnailSvg: buildThumbnailSvg(state.pattern),
    };
  });
}

export async function renameMolde(
  moldeId: string,
  userId: string,
  name: string,
): Promise<boolean> {
  const renamed = await withMolde(moldeId, userId, (state) => {
    state.pattern = { ...state.pattern, name };
  });
  return renamed !== null;
}

export async function deleteMolde(
  moldeId: string,
  userId: string,
): Promise<boolean> {
  const res = await prisma.molde.deleteMany({
    where: { id: moldeId, userId },
  });
  return res.count > 0;
}
```

```ts
// lib/db/molde-wallet.ts
import { prisma } from "./prisma";
import { lockMolde, saveMolde, TX_OPTS } from "./molde";
import { lockWallet, saveWallet, saveLedger } from "./wallet";
import type { LedgerEntry, Wallet } from "@/lib/billing/wallet";
import type { MoldeState } from "@/lib/types";

// Locks BOTH rows in the SAME transaction, atomically — Molde ALWAYS first,
// then Wallet, in that fixed order. This is the only function in the codebase
// allowed to hold both locks at once; withWallet/withMolde alone are never
// combined with this in the same request, so the lock order can never invert
// and deadlock. Don't add a second combined-lock helper without preserving
// this order.
//
// If the molde doesn't exist or isn't the caller's, returns null WITHOUT
// locking the wallet — routes turn that into moldeNotFound().
export async function withMoldeAndWallet<T>(
  moldeId: string,
  userId: string,
  fn: (
    state: MoldeState,
    wallet: Wallet,
  ) =>
    | { result: T; entries: LedgerEntry[] }
    | Promise<{ result: T; entries: LedgerEntry[] }>,
): Promise<T | null> {
  return prisma.$transaction(async (tx) => {
    const state = await lockMolde(tx, moldeId, userId);
    if (!state) return null;
    const wallet = await lockWallet(tx, userId);
    const { result, entries } = await fn(state, wallet);
    await saveMolde(tx, moldeId, state);
    await saveWallet(tx, userId, wallet);
    await saveLedger(tx, userId, entries);
    return result;
  }, TX_OPTS);
}
```

```ts
// app/api/moldes/[id]/route.ts
import { NextResponse } from "next/server";
import { requireUser } from "@/lib/auth/guard";
import {
  loadMolde,
  renameMolde,
  deleteMolde,
  moldeNotFound,
} from "@/lib/db/molde";
import { loadWallet } from "@/lib/db/wallet";
import { composeEditorState } from "@/lib/api/compose";

type Ctx = { params: Promise<{ id: string }> };

export async function GET(_request: Request, { params }: Ctx) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;
  const { id } = await params;

  const [moldeState, wallet] = await Promise.all([
    loadMolde(id, gate.id),
    loadWallet(gate.id),
  ]);
  if (!moldeState) return moldeNotFound();
  return NextResponse.json(composeEditorState(moldeState, wallet, new Date()));
}

export async function PATCH(request: Request, { params }: Ctx) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;
  const { id } = await params;

  let body: { name?: unknown };
  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { status: "error", error: "Corpo inválido." },
      { status: 400 },
    );
  }
  const name = typeof body.name === "string" ? body.name.trim() : "";
  if (!name) {
    return NextResponse.json(
      { status: "error", error: "Nome obrigatório." },
      { status: 400 },
    );
  }
  const ok = await renameMolde(id, gate.id, name);
  if (!ok) return moldeNotFound();
  return NextResponse.json({ status: "ok" });
}

export async function DELETE(_request: Request, { params }: Ctx) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;
  const { id } = await params;

  const ok = await deleteMolde(id, gate.id);
  if (!ok) return moldeNotFound();
  return NextResponse.json({ status: "ok" });
}
```

```ts
// app/api/moldes/[id]/generate/route.ts
import { NextResponse } from "next/server";
import { getTranslations } from "next-intl/server";
import { withMoldeAndWallet } from "@/lib/db/molde-wallet";
import { moldeNotFound } from "@/lib/db/molde";
import { addMessage, applyGarmentSpec, bumpVersion } from "@/lib/editor/state";
import { debitWallet, refundWallet } from "@/lib/billing/wallet";
import type { LedgerEntry } from "@/lib/billing/wallet";
import { composeEditorState } from "@/lib/api/compose";
import { COSTS } from "@/lib/types";
import { proposeGarment } from "@/lib/ai";
import { requireUser } from "@/lib/auth/guard";

type Ctx = { params: Promise<{ id: string }> };

export async function POST(request: Request, { params }: Ctx) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;
  const { id } = await params;

  let body: { message?: unknown; answering?: unknown };
  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { status: "error", error: "Corpo inválido." },
      { status: 400 },
    );
  }
  const trimmed = typeof body.message === "string" ? body.message.trim() : "";
  if (!trimmed) {
    return NextResponse.json(
      { status: "error", error: "Mensagem obrigatória." },
      { status: 400 },
    );
  }
  const answering = body.answering === true;
  const cost = COSTS.adjust;
  const t = await getTranslations("PatternData");
  const now = new Date();

  const result = await withMoldeAndWallet(
    id,
    gate.id,
    async (
      state,
      wallet,
    ): Promise<{ result: NextResponse; entries: LedgerEntry[] }> => {
      if (wallet.balance < cost) {
        return {
          result: NextResponse.json(
            {
              status: "insufficient_credits",
              required: cost,
              balance: wallet.balance,
            },
            { status: 402 },
          ),
          entries: [],
        };
      }

      addMessage(state, {
        id: `msg-${Date.now()}-u`,
        author: "user",
        text: trimmed,
        createdAt: Date.now(),
      });
      const entries: LedgerEntry[] = [debitWallet(wallet, cost, "generate")];

      try {
        const decision = await proposeGarment(state, answering);

        if (decision.decision === "ask") {
          addMessage(state, {
            id: `msg-${Date.now()}-ai`,
            author: "ai",
            text:
              decision.assistantMessage ||
              "Pra acertar a modelagem, me responde:",
            questions: decision.questions,
            createdAt: Date.now(),
          });
          entries.push(refundWallet(wallet, cost, "generate-ask"));
          return {
            result: NextResponse.json({
              status: "clarify",
              state: composeEditorState(state, wallet, now),
            }),
            entries,
          };
        }

        if (decision.decision === "decline") {
          addMessage(state, {
            id: `msg-${Date.now()}-ai`,
            author: "ai",
            text:
              decision.assistantMessage ||
              decision.declineReason ||
              "Não consegui aplicar esse pedido.",
            createdAt: Date.now(),
          });
          entries.push(refundWallet(wallet, cost, "generate-declined"));
          return {
            result: NextResponse.json({
              status: "declined",
              state: composeEditorState(state, wallet, now),
            }),
            entries,
          };
        }

        applyGarmentSpec(state, decision.spec, t);
        bumpVersion(state);
        const conflict = state.conflict;
        addMessage(state, {
          id: `msg-${Date.now()}-ai`,
          author: "ai",
          text: conflict
            ? `${decision.assistantMessage} ⚠️ ${conflict.description}`
            : decision.assistantMessage,
          chips: [{ label: `−${cost} crédito`, tone: "credit" }],
          tone: conflict ? "conflict" : undefined,
          createdAt: Date.now(),
        });
        return {
          result: NextResponse.json({
            status: conflict ? "conflict" : "ok",
            state: composeEditorState(state, wallet, now),
          }),
          entries,
        };
      } catch {
        entries.push(refundWallet(wallet, cost, "generate-error"));
        return {
          result: NextResponse.json({
            status: "error",
            state: composeEditorState(state, wallet, now),
          }),
          entries,
        };
      }
    },
  );
  return result ?? moldeNotFound();
}
```

```ts
  addMessage,
  bumpVersion,

  setConflict,
} from "@/lib/editor/state";
import { requireUser } from "@/lib/auth/guard";

type Ctx = { params: Promise<{ id: string }> };

export async function POST(request: Request, { params }: Ctx) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;
  const { id } = await params;

  let action: "fix" | "ignore" = "ignore";
  try {
    const body = (await request.json()) as { action?: unknown };
    if (body.action === "fix") action = "fix";
  } catch {
    return NextResponse.json(
      { status: "error", error: "Corpo inválido." },
      { status: 400 },
    );

        chips: [{ label: "correção gratuita", tone: "neutral" }],
        createdAt: Date.now(),
      });
    } else {
      setConflict(state, null);
    }
    return state;
  });
  if (!moldeState) return moldeNotFound();

  const wallet = await loadWallet(gate.id);
  return NextResponse.json({

    state: composeEditorState(moldeState, wallet, new Date()),
```

```ts
// app/api/billing/purchase/route.ts
import { NextResponse } from "next/server";
import { withWallet } from "@/lib/db/wallet";
import { loadMolde, moldeNotFound } from "@/lib/db/molde";
import { applyPurchase } from "@/lib/billing/wallet";
import { buildCredits, composeEditorState } from "@/lib/api/compose";
import { requireUser } from "@/lib/auth/guard";

export async function POST(request: Request) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;

  const { type, moldeId } = (await request.json()) as {
    type: "topup" | "subscribe";
    moldeId?: string;
  };
  const now = new Date();

  const wallet = await withWallet(gate.id, (wallet) => {
    const entry = applyPurchase(wallet, type, now);
    return { result: wallet, entries: [entry] };
  });

  if (typeof moldeId === "string") {
    const moldeState = await loadMolde(moldeId, gate.id);
    if (!moldeState) return moldeNotFound();
    return NextResponse.json({
      status: "ok",
      state: composeEditorState(moldeState, wallet, now),
    });
  }

  return NextResponse.json({
    status: "ok",
    credits: buildCredits(wallet, now),
  });
}
```

```tsx
// app/molde/[id]/page.tsx
import { redirect } from "next/navigation";
import { auth } from "@/auth";
import { loadMolde } from "@/lib/db/molde";
import { EditorShell } from "@/components/editor/EditorShell";

export default async function MoldePage({
  params,
}: {
  params: Promise<{ id: string }>;
}) {
  const session = await auth();
  if (!session?.user?.email || !session.user.id) redirect("/login");
  const { id } = await params;

  const moldeState = await loadMolde(id, session.user.id);
  if (!moldeState) redirect("/");

  return (
    <div className="flex min-h-0 flex-1 flex-col">
      <EditorShell userEmail={session.user.email} moldeId={id} />
    </div>
  );
}
```
