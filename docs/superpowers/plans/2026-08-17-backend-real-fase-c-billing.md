# Backend real — Fase C (billing/créditos reais) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the mock `credits` field living inside the `Molde` JSON blob with a
real `Wallet` + `Transaction` ledger in Postgres, plus an automatic cron-based
subscription renewal — the last piece of the "real backend" roadmap (A auth, B
persistence, C billing).

**Architecture:** New `Wallet`/`Transaction` Prisma models, mirroring the existing
`Molde` pattern (pure DB-free logic in `lib/billing/wallet.ts`, thin Prisma
repository in `lib/db/wallet.ts`). The two endpoints that debit credit atomically
with a molde mutation (`generate`, `graduate`) use a new `withMoldeAndWallet`
helper that locks both rows in one transaction, **always** `Molde` first then
`Wallet`, to make deadlock structurally impossible. The client/store/components
are untouched — `EditorState.credits` keeps its exact current shape, now composed
server-side from the real `Wallet` via `composeEditorState`.

**Tech Stack:** Next.js 16 route handlers, Prisma 6 + Postgres, Vitest, `tsx` for
the standalone cron script (same pattern as `prisma/seed.ts`).

## Global Constraints

- Spec: `docs/superpowers/specs/2026-08-17-backend-real-fase-c-billing-design.md`.
- No real payment gateway — "purchase"/"subscribe" keep simulating instant success.
- Balance resets to `total` on renewal (doesn't accumulate).
- No new UI — `Topbar`/`PaywallModal` and every other component stay byte-for-byte
  unchanged; only the server-side source of `credits` changes.
- Lock order invariant: whenever both `Molde` and `Wallet` rows are locked in the
  same transaction, `Molde` is always locked first, `Wallet` second. Only
  `withMoldeAndWallet` (`lib/db/molde-wallet.ts`) is allowed to lock both; no other
  function may ever acquire both locks.
- Commit messages: English, Conventional Commits (`type(scope): summary`).
- Verify with `npm run build` (not bare `tsc`) and `npm test`; lint scoped to
  `app lib components` (`npx eslint app lib components`).
- No backfill migration for existing dev-seeded wallets — first access after
  deploy lazily seeds a fresh default wallet (documented, deliberate — see spec).

```typescript
import type { PlanId } from "@/lib/types";

export interface Wallet {
  plan: PlanId;
  planLabel: string;
  balance: number;
  total: number;
  renewsAt: Date | null;
}

export type LedgerEntryType =
  "debit" | "refund" | "topup" | "subscribe" | "renew";

export interface LedgerEntry {
  type: LedgerEntryType;
  amount: number;
  balanceAfter: number;
  reason?: string;
}

const DAY_MS = 24 * 60 * 60 * 1000;

// Every user's initial wallet (mirrors the previous mock's default: Ateliê
// plan, 320/500, "renova em 12 dias") — now a real scheduled date.
export function seedWallet(now: Date): Wallet {
  return {
    plan: "atelier",
    planLabel: "Ateliê",
    balance: 320,
    total: 500,
    renewsAt: new Date(now.getTime() + 12 * DAY_MS),
  };
}

export function debitWallet(
  wallet: Wallet,
  amount: number,
  reason: string,
): LedgerEntry {
  wallet.balance -= amount;
  return { type: "debit", amount, balanceAfter: wallet.balance, reason };
}

export function refundWallet(
  wallet: Wallet,
  amount: number,
  reason: string,
): LedgerEntry {
  wallet.balance += amount;
  return { type: "refund", amount, balanceAfter: wallet.balance, reason };
}

export function applyPurchase(
  wallet: Wallet,
  type: "topup" | "subscribe",
  now: Date,
): LedgerEntry {
  if (type === "topup") {
    wallet.balance += 100;
    return { type: "topup", amount: 100, balanceAfter: wallet.balance };
  }
  wallet.plan = "atelier";
  wallet.planLabel = "Ateliê";
  wallet.balance = 500;
  wallet.total = 500;
  wallet.renewsAt = new Date(now.getTime() + 30 * DAY_MS);
  return { type: "subscribe", amount: 500, balanceAfter: wallet.balance };
}

// Called by the renewal cron (scripts/cron-renew.ts) for wallets past renewsAt.
// Rolls forward from the *scheduled* renewsAt, not `now`, so a cron run that's
// a few hours (or days) late doesn't push the next renewal date out by the delay.
export function renewWallet(wallet: Wallet, now: Date): LedgerEntry {
  wallet.balance = wallet.total;
  const base = wallet.renewsAt ?? now;
  wallet.renewsAt = new Date(base.getTime() + 30 * DAY_MS);
  return { type: "renew", amount: wallet.total, balanceAfter: wallet.balance };
}

// Wire-facing days-remaining, replaces the old static Credits.renewsInDays.
export function renewsInDays(wallet: Wallet, now: Date): number {
  if (!wallet.renewsAt) return 0;
  return Math.max(
    0,
    Math.ceil((wallet.renewsAt.getTime() - now.getTime()) / DAY_MS),
  );
}
```

```typescript
import { Prisma } from "@prisma/client";
import { prisma } from "./prisma";
import { seedMoldeState } from "@/lib/editor/state";
import type { MoldeState } from "@/lib/types";

type Tx = Prisma.TransactionClient;

// The timeout needs to cover the AI call (proposeGarment ~5–12s), which runs
// INSIDE the transaction — Prisma's 5s default would time out. Holding the
// lock during the AI call is intentional: it's what serializes the user's requests.
export const TX_OPTS = { timeout: 30_000, maxWait: 10_000 } as const;

function toState(data: Prisma.JsonValue): MoldeState {
  return data as unknown as MoldeState;
}

// Materializes the seed if missing (INSERT ... ON CONFLICT DO NOTHING), then
// locks and reads the row (SELECT ... FOR UPDATE). Exported so
// lib/db/molde-wallet.ts can lock this row inside its own combined transaction
// without duplicating the SQL.
export async function lockMolde(tx: Tx, userId: string): Promise<MoldeState> {
  const seed = JSON.stringify(seedMoldeState());
  await tx.$executeRaw`
    INSERT INTO "Molde" ("userId", "data", "updatedAt")
    VALUES (${userId}, ${seed}::jsonb, now())
    ON CONFLICT ("userId") DO NOTHING`;
  const rows = await tx.$queryRaw<{ data: Prisma.JsonValue }[]>`
    SELECT "data" FROM "Molde" WHERE "userId" = ${userId} FOR UPDATE`;
  return toState(rows[0].data);
}

export async function saveMolde(
  tx: Tx,
  userId: string,
  state: MoldeState,
): Promise<void> {
  await tx.$executeRaw`
    UPDATE "Molde" SET "data" = ${JSON.stringify(state)}::jsonb, "updatedAt" = now()
    WHERE "userId" = ${userId}`;
}

// load → fn(state) → save, inside an interactive transaction with a row lock.
export async function withMolde<T>(
  userId: string,
  fn: (state: MoldeState) => T | Promise<T>,
): Promise<T> {
  return prisma.$transaction(async (tx) => {
    const state = await lockMolde(tx, userId);
    const result = await fn(state);
    await saveMolde(tx, userId, state);
    return result;
  }, TX_OPTS);
}

// Read-only fetch for the GET: returns seedMoldeState() without writing if there's no row.
export async function loadMolde(userId: string): Promise<MoldeState> {
  const row = await prisma.molde.findUnique({ where: { userId } });
  return row ? toState(row.data) : seedMoldeState();
}

// Resets the user's molde to the blank molde and persists it.
export async function resetMolde(userId: string): Promise<MoldeState> {
  const state = seedMoldeState();
  const data = state as unknown as Prisma.InputJsonValue;
  await prisma.molde.upsert({
    where: { userId },
    create: { userId, data },
    update: { data },
  });
  return state;
}
```

```typescript
import { Prisma } from "@prisma/client";
import { prisma } from "./prisma";
import { seedWallet } from "@/lib/billing/wallet";
import type { LedgerEntry, Wallet } from "@/lib/billing/wallet";

type Tx = Prisma.TransactionClient;

export const TX_OPTS = { timeout: 30_000, maxWait: 10_000 } as const;

interface WalletRow {
  plan: string;
  planLabel: string;
  balance: number;
  total: number;
  renewsAt: Date | null;
}

function toWallet(row: WalletRow): Wallet {
  return {
    plan: row.plan as Wallet["plan"],
    planLabel: row.planLabel,
    balance: row.balance,
    total: row.total,
    renewsAt: row.renewsAt,
  };
}

// Materializes the default wallet if missing, then locks and reads the row.
// Exported so lib/db/molde-wallet.ts can lock this row inside its own combined
// transaction without duplicating the SQL.
export async function lockWallet(tx: Tx, userId: string): Promise<Wallet> {
  const seed = seedWallet(new Date());
  await tx.$executeRaw`
    INSERT INTO "Wallet" ("userId", "plan", "planLabel", "balance", "total", "renewsAt", "updatedAt")
    VALUES (${userId}, ${seed.plan}, ${seed.planLabel}, ${seed.balance}, ${seed.total}, ${seed.renewsAt}, now())
    ON CONFLICT ("userId") DO NOTHING`;
  const rows = await tx.$queryRaw<WalletRow[]>`
    SELECT "plan", "planLabel", "balance", "total", "renewsAt"
    FROM "Wallet" WHERE "userId" = ${userId} FOR UPDATE`;
  return toWallet(rows[0]);
}

export async function saveWallet(
  tx: Tx,
  userId: string,
  wallet: Wallet,
): Promise<void> {
  await tx.$executeRaw`
    UPDATE "Wallet"
    SET "plan" = ${wallet.plan}, "planLabel" = ${wallet.planLabel},
        "balance" = ${wallet.balance}, "total" = ${wallet.total},
        "renewsAt" = ${wallet.renewsAt}, "updatedAt" = now()
    WHERE "userId" = ${userId}`;
}

export async function saveLedger(
  tx: Tx,
  userId: string,
  entries: LedgerEntry[],
): Promise<void> {
  if (entries.length === 0) return;
  await tx.transaction.createMany({
    data: entries.map((e) => ({
      userId,
      type: e.type,
      amount: e.amount,
      balanceAfter: e.balanceAfter,
      reason: e.reason ?? null,
    })),
  });
}

export async function withWallet<T>(
  userId: string,
  fn: (
    wallet: Wallet,
  ) =>
    | { result: T; entries: LedgerEntry[] }
    | Promise<{ result: T; entries: LedgerEntry[] }>,
): Promise<T> {
  return prisma.$transaction(async (tx) => {
    const wallet = await lockWallet(tx, userId);
    const { result, entries } = await fn(wallet);
    await saveWallet(tx, userId, wallet);
    await saveLedger(tx, userId, entries);
    return result;
  }, TX_OPTS);
}

export async function loadWallet(userId: string): Promise<Wallet> {
  const row = await prisma.wallet.findUnique({ where: { userId } });
  return row ? toWallet(row) : seedWallet(new Date());
}

export async function resetWallet(userId: string): Promise<Wallet> {
  const wallet = seedWallet(new Date());
  await prisma.wallet.upsert({
    where: { userId },
    create: { userId, ...wallet },
    update: { ...wallet },
  });
  return wallet;
}
```

import { prisma } from "./prisma";
import { lockMolde, saveMolde, TX_OPTS } from "./molde";
import { lockWallet, saveWallet, saveLedger } from "./wallet";
import type { LedgerEntry, Wallet } from "@/lib/billing/wallet";
import type { MoldeState } from "@/lib/types";

// Locks BOTH rows in the SAME transaction, atomically — Molde ALWAYS first,
// then Wallet, in that fixed order. This is the only function in the codebase
// allowed to hold both locks at once; withWallet/withMolde alone are never
// combined with this in the same request, so the lock order can never invert
// and deadlock. Don't add a second combined-lock helper without preserving
// this order.
export async function withMoldeAndWallet<T>(
  userId: string,
  fn: (
    state: MoldeState,
    wallet: Wallet,
  ) =>
    | { result: T; entries: LedgerEntry[] }
    | Promise<{ result: T; entries: LedgerEntry[] }>,
): Promise<T> {
  return prisma.$transaction(async (tx) => {
    const state = await lockMolde(tx, userId);
    const wallet = await lockWallet(tx, userId);
    const { result, entries } = await fn(state, wallet);
    await saveMolde(tx, userId, state);
    await saveWallet(tx, userId, wallet);
    await saveLedger(tx, userId, entries);
    return result;
  }, TX_OPTS);
}

import { NextResponse } from "next/server";
import { getTranslations } from "next-intl/server";
import { withMoldeAndWallet } from "@/lib/db/molde-wallet";
import { addMessage, applyGarmentSpec, bumpVersion } from "@/lib/editor/state";
import { debitWallet, refundWallet } from "@/lib/billing/wallet";
import type { LedgerEntry } from "@/lib/billing/wallet";
import { composeEditorState } from "@/lib/api/compose";
import { COSTS } from "@/lib/types";
import { proposeGarment } from "@/lib/ai";
import { requireUser } from "@/lib/auth/guard";

export async function POST(request: Request) {
  const gate = await requireUser();
  if (gate instanceof NextResponse) return gate;

  let body: { message?: unknown; answering?: unknown };
  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { status: "error", error: "Corpo inválido." },
      { status: 400 },
    );
  }
  const trimmed = typeof body.message === "string" ? body.message.trim() : "";
  if (!trimmed) {
    return NextResponse.json(
      { status: "error", error: "Mensagem obrigatória." },
      { status: 400 },
    );
  }
  const answering = body.answering === true;
  const cost = COSTS.adjust;
  const t = await getTranslations("PatternData");
  const now = new Date();

  return withMoldeAndWallet(gate.id, async (state, wallet) => {
    if (wallet.balance < cost) {
      return {
        result: NextResponse.json(
          {
            status: "insufficient_credits",
            required: cost,
            balance: wallet.balance,
          },
          { status: 402 },
        ),
        entries: [],
      };
    }

    addMessage(state, {
      id: `msg-${Date.now()}-u`,
      author: "user",
      text: trimmed,
      createdAt: Date.now(),
    });
    const entries: LedgerEntry[] = [debitWallet(wallet, cost, "generate")];

    try {
      const decision = await proposeGarment(state, answering);

      if (decision.decision === "ask") {
        addMessage(state, {
          id: `msg-${Date.now()}-ai`,
          author: "ai",
          text:
            decision.assistantMessage ||
            "Pra acertar a modelagem, me responde:",
          questions: decision.questions,
          createdAt: Date.now(),
        });
        entries.push(refundWallet(wallet, cost, "generate-ask"));
        return {
          result: NextResponse.json({
            status: "clarify",
            state: composeEditorState(state, wallet, now),
          }),
          entries,
        };
      }

      if (decision.decision === "decline") {
        addMessage(state, {
          id: `msg-${Date.now()}-ai`,
          author: "ai",
          text:
            decision.assistantMessage ||
            decision.declineReason ||
            "Não consegui aplicar esse pedido.",
          createdAt: Date.now(),
        });
        entries.push(refundWallet(wallet, cost, "generate-declined"));
        return {
          result: NextResponse.json({
            status: "declined",
            state: composeEditorState(state, wallet, now),
          }),
          entries,
        };
      }

      applyGarmentSpec(state, decision.spec, t);
      bumpVersion(state);
      const conflict = state.conflict;
      addMessage(state, {
        id: `msg-${Date.now()}-ai`,
        author: "ai",
        text: conflict
          ? `${decision.assistantMessage} ⚠️ ${conflict.description}`
          : decision.assistantMessage,
        chips: [{ label: `−${cost} crédito`, tone: "credit" }],
        tone: conflict ? "conflict" : undefined,
        createdAt: Date.now(),
      });
      return {
        result: NextResponse.json({
          status: conflict ? "conflict" : "ok",
          state: composeEditorState(state, wallet, now),
        }),
        entries,
      };
    } catch {
      entries.push(refundWallet(wallet, cost, "generate-error"));
      return {
        result: NextResponse.json({
          status: "error",
          state: composeEditorState(state, wallet, now),
        }),
        entries,
      };
    }
  });
}

import { PrismaClient } from "@prisma/client";
import { withWallet } from "../lib/db/wallet";
import { renewWallet } from "../lib/billing/wallet";

const prisma = new PrismaClient();

async function main() {
  const now = new Date();
  const due = await prisma.wallet.findMany({
    where: { renewsAt: { lte: now } },
    select: { userId: true },
  });

  for (const { userId } of due) {
    await withWallet(userId, (wallet) => {
      const entry = renewWallet(wallet, now);
      return { result: undefined, entries: [entry] };
    });
  }

  console.log(

  );
}

main()
  .then(() => process.exit(0))
  .catch((e) => {
    console.error("[cron-renew] failed:", e);
    process.exit(1);
  })
  .finally(() => prisma.$disconnect());
